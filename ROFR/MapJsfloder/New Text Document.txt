function Itdagismap() {
    var map = null;
    var location = null;
    if ($("#selecteddistrict").val() == "s") {
        alert('Please select itda');

    }
    else {


        var hidField1 = $("#selecteddistrict").val();

        var hidField2 = "";
        var hidField3 = "";
        var hidField4 = "";
        var hidField5 = "1";
        $.ajax(
       {
           type: 'POST',
           contentType: 'application/json; charset=utf-8',
           url: '/api/ITDA/ItdaLatlongslist',
           data: "{'ITDA':'" + hidField1 + "', 'MANDAL':'" + hidField2 + "', 'VILLAGE':'" + hidField3 + "', 'Category':'" + hidField4 + "', 'Type':'" + hidField5 + "'}",
           dataType: "json",
           success: function (response) {
               console.log(JSON.stringify(response));
               var dist = response;
               var LAT = '17.998426';
               var LONG = '82.739254';
               location = new google.maps.LatLng(LAT, LONG);
               map = new google.maps.Map(document.getElementById('map'), {
                   zoom: 10,
                   center: location,
                   mapTypeId: google.maps.MapTypeId.ROADMAP  //HYBRID
               });
               var bounds = new google.maps.LatLngBounds();
               arr = [];
               for (var i = 0; i < dist.itdalist.length; i++) {

                   arr.push(new google.maps.LatLng(
                           parseFloat(dist.itdalist[i].LATTITUDE),
                         parseFloat(dist.itdalist[i].LONGITUDE)
                    ));
                   bounds.extend(arr[arr.length - 1])


               }

               if (dist.itdalist.length >= 3000) {

                   polygons.push(new google.maps.Polygon({
                       paths: arr,
                       strokeColor: '#FF0000',
                       strokeOpacity: 0.8,
                       strokeWeight: 3,
                       fillColor: '',
                       fillOpacity: 0
                   }));
                   polygons[polygons.length - 1].setMap(map);
                   map.fitBounds(bounds);
               }
               else {
                   polygons.push(new google.maps.Polygon({
                       paths: arr,
                       strokeColor: '#FF0000',
                       strokeOpacity: 0.8,
                       strokeWeight: 3,
                       fillColor: '',
                       fillOpacity: 0
                   }));
                   polygons[polygons.length - 1].setMap(map);
                   map.fitBounds(bounds);
               }



           },
           error: function (result) {
               alert("Error");
           }

       });
    }


}

function Itdas() {
    var hidField1 = "";
    var hidField2 = "";
    var hidField3 = "";
    var hidField4 = "";
    var hidField5 = "6";
    $.ajax(
        {
            type: "POST",
            contentType: "application/json; charset=utf-8",
            url: '/api/ITDA/ITDAMANDALVILLAGEdropdowns',
            data: "{'ITDA':'" + hidField1 + "', 'MANDAL':'" + hidField2 + "', 'VILLAGE':'" + hidField3 + "', 'Category':'" + hidField4 + "', 'Type':'" + hidField5 + "'}",
            dataType: 'json',
            success: function (response) {
                console.log(JSON.stringify(response));
                var Itda = response;
                for (var i = 0; i < Itda.itdalist.length; i++) {
                    var opt1 = new Option(Itda.itdalist[i].ITDA_NAME);
                    $("#selecteddistrict").append(opt1);
                }
            },
            error: function (result) {
                alert("Error");
            }
        });
}
